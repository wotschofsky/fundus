cmake_minimum_required(VERSION 3.18)
project(vault_audio C)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BUILD_PROGRAMS OFF CACHE BOOL "" FORCE)
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(BUILD_CXXLIBS OFF CACHE BOOL "" FORCE)
set(WITH_OGG OFF CACHE BOOL "" FORCE)
set(INSTALL_MANPAGES OFF CACHE BOOL "" FORCE)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(NOT DEFINED FLAC_SOURCE_DIR)
 set(FLAC_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../.cache/native/flac-1.5.0")
endif()
# Bionic exposes fseeko as an alias of fseeko64 on 32-bit API 24+;
# a symbol-only CMake probe incorrectly reports it missing.
if(ANDROID)
 set(HAVE_FSEEKO 1 CACHE INTERNAL "Bionic header alias" FORCE)
 add_compile_definitions(_FILE_OFFSET_BITS=64)
endif()
add_subdirectory("${FLAC_SOURCE_DIR}" flac EXCLUDE_FROM_ALL)
if(ANDROID)
 target_compile_definitions(FLAC PRIVATE HAVE_FSEEKO)
endif()
add_library(vault_audio STATIC vault_audio.c)
target_include_directories(vault_audio PUBLIC include)
target_link_libraries(vault_audio PUBLIC FLAC)
